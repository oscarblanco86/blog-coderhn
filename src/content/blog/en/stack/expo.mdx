## 3. Reading the accelerometer

```tsx
import { Accelerometer } from 'expo-sensors';
import { useEffect, useState } from 'react';

export function useAcceleration(intervalMs = 20) {
	const [sample, setSample] = useState({ x: 0, y: 0, z: 0 });
	useEffect(() => {
		Accelerometer.setUpdateInterval(intervalMs);
		const sub = Accelerometer.addListener(setSample);
		return () => sub.remove();
	}, [intervalMs]);
	return sample;
}
```
