## 3. My own server in TypeScript

```ts
// TODO: validate against the current version of @modelcontextprotocol/sdk
import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js';
import { StdioServerTransport } from '@modelcontextprotocol/sdk/server/stdio.js';
import { z } from 'zod';
import { karvonen } from 'run-science'; // TODO: real import

const server = new McpServer({ name: 'run-science', version: '0.1.0' });

server.tool(
	'karvonen_zone',
	'Computes the target HR for a given intensity',
	{ restHr: z.number(), maxHr: z.number(), intensity: z.number().min(0).max(1) },
	async ({ restHr, maxHr, intensity }) => ({
		content: [{ type: 'text', text: String(Math.round(karvonen(restHr, maxHr, intensity))) }],
	}),
);

await server.connect(new StdioServerTransport());
```
