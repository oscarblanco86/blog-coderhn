## 2. The blog's architecture

```text
src/
├── content/blog/es/     → Spanish posts (MDX)
├── content/blog/en/     → English posts, same slug = translation
├── content.config.ts    → Zod schema for the frontmatter
├── i18n/ui.ts           → UI strings and localized paths
├── lib/posts.ts         → getPosts(lang): hides drafts in production
├── views/               → pages shared by both languages
└── components/          → Terminal, EngineeringSummary, ThemeToggle, LangSwitch
```

## 3. Drafts that never reach production

```ts
// src/lib/posts.ts
const isVisible = (post: Post) => import.meta.env.DEV || !post.data.draft;

export async function getPosts(lang: Lang) {
	const posts = await getCollection('blog', (p) => postLang(p) === lang && isVisible(p));
	return posts.sort((a, b) => b.data.pubDate.valueOf() - a.data.pubDate.valueOf());
}
```
