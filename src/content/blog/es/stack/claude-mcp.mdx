## 3. Mi propio servidor en TypeScript

```ts
// TODO: validar contra la versión actual de @modelcontextprotocol/sdk
import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js';
import { StdioServerTransport } from '@modelcontextprotocol/sdk/server/stdio.js';
import { z } from 'zod';
import { karvonen } from 'run-science'; // TODO: import real

const server = new McpServer({ name: 'run-science', version: '0.1.0' });

server.tool(
	'zona_karvonen',
	'Calcula la FC objetivo para una intensidad dada',
	{ restHr: z.number(), maxHr: z.number(), intensity: z.number().min(0).max(1) },
	async ({ restHr, maxHr, intensity }) => ({
		content: [{ type: 'text', text: String(Math.round(karvonen(restHr, maxHr, intensity))) }],
	}),
);

await server.connect(new StdioServerTransport());
```
