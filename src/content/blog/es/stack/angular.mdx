## 3. El mismo componente, moderno

```ts
import { Component, computed, input, signal } from '@angular/core';

@Component({
	selector: 'app-athletes',
	template: `
		<input (input)="query.set($any($event.target).value)" placeholder="Buscar" />
		@for (a of filtered(); track a.id) {
			<p>{{ a.name }}</p>
		} @empty {
			<p>Sin resultados</p>
		}
	`,
})
export class AthletesComponent {
	athletes = input.required<{ id: number; name: string }[]>();
	query = signal('');
	filtered = computed(() =>
		this.athletes().filter((a) => a.name.toLowerCase().includes(this.query().toLowerCase())),
	);
}
```
